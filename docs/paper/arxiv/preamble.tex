% Unicode-capable fonts: the source uses em dashes, minus signs and multiplication
% signs, and the default Latin Modern encoding drops some of them silently.
\usepackage{fontspec}
% arXiv's XeLaTeX requires filename lookup for TeX Live fonts.
\setmainfont{lmroman10-regular.otf}[
  BoldFont=lmroman10-bold.otf,
  ItalicFont=lmroman10-italic.otf,
  BoldItalicFont=lmroman10-bolditalic.otf]
\setmonofont{lmmono10-regular.otf}[
  ItalicFont=lmmono10-italic.otf,Scale=MatchLowercase]

% arXiv preprint preamble. Single column, generous margins, no anonymity.
\usepackage{booktabs}
\usepackage{array}
\usepackage[table]{xcolor}
\usepackage{microtype}
\usepackage{amsmath}
\usepackage[hidelinks]{hyperref}

% Tables in this paper are numeric and dense; keep them readable.
\renewcommand{\arraystretch}{1.15}
\setlength{\tabcolsep}{5pt}

% pandoc emits longtable for pipe tables; make them small enough to fit.
\AtBeginEnvironment{longtable}{\small}
\AtBeginEnvironment{quote}{\small}

% Blockquotes are used for status notes, not epigraphs.
\usepackage{etoolbox}
\usepackage{framed}
\colorlet{shadecolor}{gray!8}
\renewenvironment{quote}{\begin{shaded}\small}{\end{shaded}}

\usepackage{titlesec}
\titleformat{\section}{\large\bfseries}{\thesection}{0.6em}{}
\titleformat{\subsection}{\normalsize\bfseries}{\thesubsection}{0.6em}{}

% Figures and captions
\usepackage{caption}
\captionsetup{font=small,labelfont=bf}
% Keep figures near their first mention
\usepackage{float}
\floatplacement{figure}{H}
